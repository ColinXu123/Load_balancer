30bool Config::load(const std::string& path) {
31 std::ifstream in(path);
32 if (!in) return false;
33 sourcePath_ = path;
34
35 std::string line;
36 while (std::getline(in, line)) {
37 const auto hash = line.find('#');
38 if (hash != std::string::npos) line = line.substr(0, hash);
39 line = trim(line);
40 if (line.empty()) continue;
41
42 const auto eq = line.find('=');
43 if (eq == std::string::npos) continue;
44 const std::string key = lower(trim(line.substr(0, eq)));
45 const std::string value = trim(line.substr(eq + 1));
46 if (key.empty()) continue;
47
48 if (key == "blocked_range") {
49 if (!value.empty()) blockedRanges_.push_back(value);
50 } else {
51 values_[key] = value;
52 }
53 }
54 return true;
55}
