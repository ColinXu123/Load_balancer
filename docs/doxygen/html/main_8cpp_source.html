86Options parseArgs(int argc, char** argv) {
87 Options o;
88 for (int i = 1; i < argc; ++i) {
89 const std::string a = argv[i];
90 auto value = [&](const std::string& flag) -> std::string {
91 if (i + 1 >= argc) {
92 std::cerr << "Missing value for " << flag << "\n";
93 std::exit(2);
94 }
95 return argv[++i];
96 };
97 if (a == "-c" || a == "--config") o.configPath = value(a);
98 else if (a == "-s" || a == "--servers") o.servers = value(a);
99 else if (a == "-t" || a == "--cycles") o.cycles = value(a);
100 else if (a == "--seed") o.seed = value(a);
101 else if (a == "--log") o.logPath = value(a);
102 else if (a == "-v" || a == "--verbose") o.verbose = 1;
103 else if (a == "-q" || a == "--quiet") o.verbose = 0;
104 else if (a == "--switch") o.useSwitch = 1;
105 else if (a == "--single") o.useSwitch = 0;
106 else if (a == "--no-color") o.noColor = true;
107 else if (a == "--set") o.overrides.push_back(value(a));
108 else if (a == "-h" || a == "--help") { printUsage(argv[0]); std::exit(0); }
109 else {
110 std::cerr << "Unknown option: " << a << "\n";
111 printUsage(argv[0]);
112 std::exit(2);
113 }
114 }
115 return o;
116}

125long obtainPositive(const std::string& given, const std::string& prompt, long fallback) {
126 std::string text = given;
127 if (text.empty() && isatty(STDIN_FILENO)) {
128 std::cout << prompt << " [" << fallback << "]: " << std::flush;
129 std::getline(std::cin, text);
130 }
131 if (text.empty()) return fallback;
132 try {
133 const long v = std::stol(text);
134 if (v > 0) return v;
135 } catch (...) {
136 }
137 std::cerr << "Invalid value '" << text << "', using " << fallback << "\n";
138 return fallback;
139}

149int main(int argc, char** argv) {
150 const Options opt = parseArgs(argc, argv);
151
152 Config config;
153 const bool configLoaded = config.load(opt.configPath);
154 if (!configLoaded) {
155 std::cerr << "Warning: could not read '" << opt.configPath
156 << "', using built-in defaults.\n";
157 }
158 for (const auto& kv : opt.overrides) {
159 const auto eq = kv.find('=');
160 if (eq == std::string::npos) {
161 std::cerr << "Ignoring --set '" << kv << "' (expected KEY=VALUE).\n";
162 continue;
163 }
164 config.set(kv.substr(0, eq), kv.substr(eq + 1));
165 }
166 if (!opt.seed.empty()) config.set("seed", opt.seed);
167
168 const long servers = obtainPositive(opt.servers, "Number of web servers",
169 config.getInt("default_servers", 20));
170 const long cycles = obtainPositive(opt.cycles, "Simulation length in clock cycles",
171 config.getInt("default_cycles", 10000));
172
173 const bool verbose = opt.verbose >= 0 ? opt.verbose == 1 : config.getBool("verbose", false);
174 const bool useSwitch = opt.useSwitch >= 0 ? opt.useSwitch == 1 : config.getBool("use_switch", true);
175 const bool color = !opt.noColor && config.getBool("color", true);
176 const std::string logPath = !opt.logPath.empty() ? opt.logPath : config.getString("log_file", "");
177
178 if (!Logger::init(verbose, color, logPath)) {
179 std::cerr << "Warning: could not open log file '" << logPath << "'.\n";
180 }
181
182 IPBlocker blocker;
183 for (const auto& spec : config.blockedRanges()) {
184 if (!blocker.addRange(spec)) {
185 std::cerr << "Warning: ignoring malformed blocked_range '" << spec << "'.\n";
186 }
187 }
188
189 RequestGenerator generator(config, blocker, static_cast<unsigned>(config.getInt("seed", 0)));
190 Switch net(config, generator, blocker, static_cast<int>(servers), useSwitch);
191
192 const int perServer = static_cast<int>(config.getInt("initial_requests_per_server", 100));
193 net.fillInitialQueues(perServer);
194
195 // ---- Initial configuration ----------------------------------------------
196 auto heading = [](const std::string& title) {
197 Logger::always("==================== " + title + " ====================", LogColor::Bold);
198 };
199 auto item = [](const std::string& label, const std::string& value) {
200 std::ostringstream os;
201 os << " " << std::left << std::setw(26) << label << value;
202 Logger::always(os.str());
203 };
204 auto num = [](double v) {
205 std::ostringstream os;
206 os << v;
207 return os.str();
208 };
209
210 heading("LOAD BALANCER SIMULATION: INITIAL CONFIGURATION");
211 item("Config file", configLoaded ? opt.configPath : std::string("(built-in defaults)"));
212 item("Logging mode", verbose ? "verbose (every event)" : "quiet (major events + summaries)");
213 item("Routing", net.switchMode() ? "switch: S -> LB-Streaming, P -> LB-Processing"
214 : "single balancer: all jobs -> LB-Main");
215 item("Random seed", std::to_string(generator.seed()));
216 item("Starting web servers", std::to_string(servers));
217 item("Simulation length", std::to_string(cycles) + " clock cycles");
218 item("Initial queue", std::to_string(perServer) + " requests per server");
219 item("Scaling rule", "add if queue > " + std::to_string(config.getInt("scale_high_threshold", 80)) +
220 " x servers, remove if < " + std::to_string(config.getInt("scale_low_threshold", 50)) +
221 " x servers");
222 item("Scaling wait", std::to_string(config.getInt("scale_interval", 250)) + " cycles after each change");
223 item("Server limits", std::to_string(config.getInt("min_servers", 1)) + " to " +
224 std::to_string(config.getInt("max_servers", 500)) + " per balancer");
225 item("Arrival rate", num(config.getDouble("arrival_rate", 0.6)) + " requests/cycle on average");
226 item("Traffic wave", "+/-" + num(config.getDouble("traffic_wave_amplitude", 0.5) * 100) +
227 "% every " + std::to_string(config.getInt("traffic_wave_period", 25000)) + " cycles");
228 item("Traffic spikes", "x" + num(config.getDouble("spike_multiplier", 3.0)) + ", chance " +
229 num(config.getDouble("spike_chance", 0.00005)) + "/cycle");
230 item("Traffic lulls", "x" + num(config.getDouble("lull_multiplier", 0.1)) + ", chance " +
231 num(config.getDouble("lull_chance", 0.00005)) + "/cycle");
232 item("Spike/lull length", std::to_string(config.getInt("event_min_length", 1000)) + "-" +
233 std::to_string(config.getInt("event_max_length", 5000)) + " cycles");
234 item("Job mix", num(config.getDouble("streaming_ratio", 0.35) * 100) + "% streaming (S), rest processing (P)");
235 item("Job times", "P " + std::to_string(config.getInt("processing_time_min", 5)) + "-" +
236 std::to_string(config.getInt("processing_time_max", 45)) + ", S " +
237 std::to_string(config.getInt("streaming_time_min", 40)) + "-" +
238 std::to_string(config.getInt("streaming_time_max", 120)) + " cycles");
239 {
240 std::ostringstream os;
241 os << blocker.rangeSpecs().size() << " range(s):";
242 for (const auto& r : blocker.rangeSpecs()) os << " " << r;
243 item("Firewall", os.str());
244 }
245 item("Attack traffic", num(config.getDouble("attack_probability", 0.05) * 100) +
246 "% of requests from blocked ranges");
247 Logger::always("Starting state:", LogColor::Bold);
248 for (const auto& lb : net.balancers()) {
249 Logger::always(" Start " + lb->statusLine(), LogColor::Cyan);
250 }
251
252 // ---- Main loop ----------------------------------------------------------
253 heading("SIMULATION RUNNING");
254 const long statusInterval = config.getInt("status_interval", 5000);
255 for (long t = 0; t < cycles; ++t) {
256 net.step();
257 if (statusInterval > 0 && net.clock() % statusInterval == 0 && net.clock() < cycles) {
258 std::ostringstream head;
259 head << "[cycle " << net.clock() << "] status (traffic " << std::fixed
260 << std::setprecision(2) << generator.trafficLevel() << "x, "
261 << generator.trafficState() << ")";
262 Logger::always(head.str(), LogColor::Cyan);
263 for (const auto& lb : net.balancers()) {
264 Logger::always(" " + lb->statusLine(), LogColor::Cyan);
265 }
266 }
267 }
268
269 // ---- End status ---------------------------------------------------------
270 heading("END STATUS");
271 for (const auto& lb : net.balancers()) {
272 Logger::always(" End " + lb->statusLine(), LogColor::Cyan);
273 }
274 item("Traffic spikes / lulls", std::to_string(generator.spikeCount()) + " / " +
275 std::to_string(generator.lullCount()));
276 heading("FINAL SUMMARY");
277 for (const auto& lb : net.balancers()) {
278 lb->printSummary();
279 }
280 // Totals across balancers (in single mode this adds the firewall and accounting check).
281 net.printCombinedSummary();
282 Logger::always("==================================================================", LogColor::Bold);
283 Logger::close();
284 return 0;
