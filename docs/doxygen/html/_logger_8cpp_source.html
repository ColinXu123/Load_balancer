36bool Logger::init(bool verbose, bool color, const std::string& logPath) {
37 verbose_ = verbose;
38 color_ = color;
39 if (file_.is_open()) file_.close();
40 if (logPath.empty()) return true;
41 // Create any missing parent folders (e.g. "logs/") so --log just works.
42 const std::filesystem::path parent = std::filesystem::path(logPath).parent_path();
43 if (!parent.empty()) {
44 std::error_code ignored;
45 std::filesystem::create_directories(parent, ignored);
46 }
47 file_.open(logPath, std::ios::out | std::ios::trunc);
48 return file_.is_open();
49}
