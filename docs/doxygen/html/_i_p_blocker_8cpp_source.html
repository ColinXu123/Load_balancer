21bool IPBlocker::parseIP(const std::string& ip, std::uint32_t& out) {
22 std::istringstream in(trim(ip));
23 std::uint32_t value = 0;
24 for (int part = 0; part < 4; ++part) {
25 int octet = -1;
26 if (!(in >> octet) || octet < 0 || octet > 255) return false;
27 value = (value << 8) | static_cast<std::uint32_t>(octet);

46bool IPBlocker::addRange(const std::string& rawSpec) {
47 const std::string spec = trim(rawSpec);
48 std::uint32_t low = 0, high = 0;
49
50 const auto slash = spec.find('/');
51 const auto dash = spec.find('-');
52 if (slash != std::string::npos) {
53 std::uint32_t base = 0;
54 int bits = -1;
55 try {
56 bits = std::stoi(spec.substr(slash + 1));
57 } catch (...) {
58 return false;
59 }
60 if (!parseIP(spec.substr(0, slash), base) || bits < 0 || bits > 32) return false;
61 const std::uint32_t mask = bits == 0 ? 0u : (0xFFFFFFFFu << (32 - bits));
62 low = base & mask;
63 high = low | ~mask;
64 } else if (dash != std::string::npos) {
65 if (!parseIP(spec.substr(0, dash), low) || !parseIP(spec.substr(dash + 1), high)) {
66 return false;
67 }
68 if (low > high) std::swap(low, high);
69 } else {
70 if (!parseIP(spec, low)) return false;
71 high = low;
72 }
