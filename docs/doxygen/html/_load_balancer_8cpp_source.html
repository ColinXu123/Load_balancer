58 queue_.push(request);

87void LoadBalancer::step() {
88 // 1. Advance every server one cycle. Completions happen almost every cycle,
89 // so they are counted (and shown per server in the summary), not logged.
90 for (auto& server : servers_) {
91 if (server->tick()) {
92 const Request& done = server->current();
93 ++stats_.processed;
94 stats_.totalServiceTime += done.time;
95 if (sampled(done)) {
96 std::ostringstream os;
97 os << "server " << server->id() << " finished request #" << done.id
98 << " " << done.jobType << " after " << done.time << " cycles";
99 Logger::event(clock_, name_, LogColor::Green, "DONE", os.str());
100 }
101 }
102 }
103
104 // 2. Retire draining servers that are now idle (their work is complete).
105 for (auto it = servers_.begin(); it != servers_.end();) {
106 if ((*it)->isDraining() && !(*it)->isBusy()) {
107 std::ostringstream os;
108 os << "server " << (*it)->id() << " drained and taken offline, active="
109 << activeServers();
110 Logger::event(clock_, name_, LogColor::Magenta, "OFFLINE", os.str());
111 ++stats_.serversDrained;
112 it = servers_.erase(it);
113 } else {
114 ++it;
115 }
116 }
117
118 // 3. Give queued work to idle servers.
119 dispatch();
120
121 // 4. Statistics for this cycle.
122 const long queued = static_cast<long>(queue_.size());
123 stats_.queueSum += queued;
124 stats_.maxQueue = std::max(stats_.maxQueue, queued);
125 stats_.serverCycles += static_cast<long>(servers_.size());
126 stats_.busyServerCycles += inFlight();
127
128 // 5. Scaling, respecting the cooldown after the last action.
129 if (cooldown_ > 0) {
130 --cooldown_;
131 } else if (evaluateScaling()) {
132 cooldown_ = scaleInterval_;
133 }
134
135 ++clock_;
136}

138void LoadBalancer::dispatch() {
139 for (auto& server : servers_) {
140 if (queue_.empty()) return;
141 if (server->canAccept()) {
142 Request next = queue_.pop();
143 stats_.totalWait += clock_ - next.createdAt;
144 ++stats_.started;
145 server->assign(next);
146 if (sampled(next)) {
147 std::ostringstream os;
148 os << "request #" << next.id << " " << next.jobType << " -> server "
149 << server->id() << " (waited " << clock_ - next.createdAt
150 << " cycles, needs " << next.time << ")";
151 Logger::event(clock_, name_, LogColor::Plain, "ASSIGN", os.str());
152 }
153 }
154 }
155}

157bool LoadBalancer::evaluateScaling() {
158 const long active = activeServers();
159 const long queued = static_cast<long>(queue_.size());
160
161 if (queued > static_cast<long>(highThreshold_) * active && active < maxServers_) {
162 addServer();
163 return true;
164 }
165 if (queued < static_cast<long>(lowThreshold_) * active && active > minServers_) {
166 removeServer();
167 return true;
168 }
169 return false;
170}

172void LoadBalancer::addServer() {
173 servers_.push_back(std::make_unique<WebServer>(nextServerId_++, clock_));
174 ++stats_.serversAdded;
175 trackServerCount();
176 std::ostringstream os;
177 os << "queue " << queue_.size() << " > " << highThreshold_ << " x "
178 << activeServers() - 1 << " -> server " << servers_.back()->id()
179 << " added, active=" << activeServers();
180 Logger::event(clock_, name_, LogColor::Yellow, "SCALE+", os.str());
181}

183void LoadBalancer::removeServer() {
184 const int before = activeServers();
185
186 // Prefer an idle server: it can leave immediately with nothing in flight.
187 auto idle = std::find_if(servers_.begin(), servers_.end(),
188 [](const std::unique_ptr<WebServer>& s) { return s->canAccept(); });
189
190 std::ostringstream os;
191 os << "queue " << queue_.size() << " < " << lowThreshold_ << " x " << before << " -> ";
192 if (idle != servers_.end()) {
193 os << "idle server " << (*idle)->id() << " removed";
194 servers_.erase(idle);
195 } else {
196 // Otherwise drain the busy server closest to finishing.
197 WebServer* pick = nullptr;
198 for (auto& s : servers_) {
199 if (s->isDraining()) continue;
200 if (!pick || s->remaining() < pick->remaining()) pick = s.get();
201 }
202 if (!pick) return;
203 pick->startDraining();
204 os << "server " << pick->id() << " draining (finishing request #"
205 << pick->current().id << ", " << pick->remaining() << " cycles left)";
206 }
207 ++stats_.serversRemoved;
208 trackServerCount();
209 os << ", active=" << activeServers();
210 Logger::event(clock_, name_, LogColor::Magenta, "SCALE-", os.str());
211}

269 Logger::always("--- " + name_ + " ---", LogColor::Bold);

295 Logger::always(row("Requests rejected by firewall",
296 std::to_string(s.blocked) + " (" +
