14Switch::Switch(const Config& config, RequestGenerator& generator, const IPBlocker& blocker,
15 int totalServers, bool useSwitch)
16 : generator_(generator), blocker_(blocker), useSwitch_(useSwitch) {
17 if (useSwitch_ && totalServers >= 2) {
18 const double share = std::clamp(config.getDouble("streaming_server_share", 0.6), 0.0, 1.0);
19 int streamServers = static_cast<int>(std::lround(totalServers * share));
20 streamServers = std::clamp(streamServers, 1, totalServers - 1);
21 lbs_.push_back(std::make_unique<LoadBalancer>("LB-Streaming", config, streamServers));
22 lbs_.push_back(std::make_unique<LoadBalancer>("LB-Processing", config,
23 totalServers - streamServers));
24 streaming_ = lbs_[0].get();
25 processing_ = lbs_[1].get();
26 } else {
27 useSwitch_ = false;
28 lbs_.push_back(std::make_unique<LoadBalancer>("LB-Main", config, totalServers));
29 streaming_ = processing_ = lbs_[0].get();
30 }
31}

33void Switch::fillInitialQueues(int perServer) {
34 for (auto& lb : lbs_) {
35 const long target = static_cast<long>(perServer) * lb->activeServers();
36 // Keep generating until the queue holds exactly the target; requests the
37 // firewall blocks along the way are still counted as blocked.
38 while (lb->stats().initialQueue < target) {
39 // In switch mode each balancer is filled with its own job type.
40 Request r = !useSwitch_ ? generator_.generate(0)
41 : generator_.generate(0, lb.get() == streaming_ ? 'S' : 'P');
42 route(r, true);
43 }
44 }
45}

47void Switch::step() {
48 const int arrivals = generator_.arrivalsThisCycle(clock_);
49 for (int i = 0; i < arrivals; ++i) {
50 route(generator_.generate(clock_));
51 }
52 for (auto& lb : lbs_) {
53 lb->step();
54 }

58void Switch::route(const Request& request, bool initial) {
59 ++generated_;
60 LoadBalancer& lb = target(request.jobType);
61 if (blocker_.isBlocked(request.ipIn)) {
62 ++blocked_;
63 lb.recordBlocked(request);
64 return;
65 }
66 lb.submit(request, initial);
