45Request RequestGenerator::generate(long cycle, char jobType) {
46 Request r;
47 r.id = nextId_++;
48 r.jobType = jobType == 'S' ? 'S' : 'P';
49 r.time = r.jobType == 'S' ? uniform(streamMin_, streamMax_)
50 : uniform(procMin_, procMax_);
51 const bool hostile = !blocker_.ranges().empty() && chance() < attackProbability_;
52 r.ipIn = hostile ? attackIP() : randomIP();
53 r.ipOut = randomIP();
54 r.createdAt = cycle;
55 return r;
56}

58int RequestGenerator::arrivalsThisCycle(long cycle) {
59 // 1. Spikes and lulls: finish the current one, or maybe start a new one.
60 if (eventLeft_ > 0) {
61 if (--eventLeft_ == 0) {
62 Logger::event(cycle, "Traffic", LogColor::Cyan, "TRAFFIC",
63 std::string(eventType_ == 'S' ? "spike" : "lull") +
64 " over, traffic back to normal");
65 eventType_ = 'N';
66 eventMult_ = 1.0;
67 }
68 } else {
69 const double r = chance();
70 const bool spike = r < spikeChance_;
71 const bool lull = !spike && r < spikeChance_ + lullChance_;
72 if (spike || lull) {
73 eventType_ = spike ? 'S' : 'L';
74 eventMult_ = spike ? spikeMultiplier_ : lullMultiplier_;
75 eventLeft_ = uniform(eventMin_, eventMax_);
76 spike ? ++spikes_ : ++lulls_;
77 std::ostringstream os;
78 os << (spike ? "SPIKE: traffic x" : "LULL: traffic x") << eventMult_
79 << " for " << eventLeft_ << " cycles";
80 Logger::event(cycle, "Traffic", spike ? LogColor::Yellow : LogColor::Cyan,
81 "TRAFFIC", os.str());
82 }
83 }
84
85 // 2. Slow wave, like busy and quiet hours of a day.
86 const double pi = 3.14159265358979323846;
87 const double wave = 1.0 + waveAmplitude_ * std::sin(2.0 * pi * static_cast<double>(cycle) /
88 static_cast<double>(wavePeriod_));
89 level_ = std::max(0.0, wave * eventMult_);
90
91 // 3. Number of arrivals this cycle, averaging arrivalRate_ x level_.
92 const double mean = arrivalRate_ * level_;
93 if (mean <= 0.0) return 0;
94 std::poisson_distribution<int> dist(mean);
95 return dist(rng_);
96}
